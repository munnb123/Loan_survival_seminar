\begin{frame}[noframenumbering]{Backup: partial likelihood}
\[
L(\beta)=\prod_{j:\,\text{default}}
\frac{\exp(\beta^{\mathsf T}x_{i_j})}
{\displaystyle\sum_{k\in\mathcal R_{y_j}(t_j)}
\exp(\beta^{\mathsf T}x_k)}
\]
\vspace{.4cm}
\qual{$\mathcal R_{y_j}(t_j)$: loans still exposed immediately before $t_j$, within the defaulting loan's origination-year stratum.}\par
\vspace{.4cm}
\qual{Schematic formula for distinct event times. The audited fit uses Efron handling of ties.}
\talknote{Optional; outside planned speaking time}{Each factor is the conditional probability of the observed defaulting identity given one event in that stratum's risk set. The baseline cancels and we combine contributions over event times. Censoring and closure affect who is in the risk set. The formula is schematic for untied events; do not imply it is the exact tied-data likelihood used by the existing software.}
\end{frame}
